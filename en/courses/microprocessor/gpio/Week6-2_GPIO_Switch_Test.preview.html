import RPi.GPIO as GPIO
import time

GPIO.setmode(GPIO.BCM)
GPIO.setup(5, GPIO.OUT)
GPIO.setup(6, GPIO.IN)

try:
    while(True):
        if GPIO.input(6):
            GPIO.output(5, True)
        else:
            GPIO.output(5, False)
finally:
    print('cleaning up')
    GPIO.cleanup()
